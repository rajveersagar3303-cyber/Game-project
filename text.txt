import os

# Create an empty board
board = [' ' for _ in range(9)]

def print_board():
    # Clears the terminal screen for a clean UI
    os.system('cls' if os.name == 'nt' else 'clear')
    print("\nTic-Tac-Toe\n")
    print(f" {board[0]} | {board[1]} | {board[2]} ")
    print("---+---+---")
    print(f" {board[3]} | {board[4]} | {board[5]} ")
    print("---+---+---")
    print(f" {board[6]} | {board[7]} | {board[8]} \n")

def check_win(player):
    # All possible winning rows, columns, and diagonals
    wins = [(0,1,2), (3,4,5), (6,7,8), (0,3,6), (1,4,7), (2,5,8), (0,4,8), (2,4,6)]
    return any(board[a] == board[b] == board[c] == player for a, b, c in wins)

current_player = 'X'

# A game has a maximum of 9 turns
for turn in range(9):
    print_board()
    
    try:
        # Ask user for input and convert to 0-based index
        move = int(input(f"Player {current_player}, choose a spot (1-9): ")) - 1
        
        if move < 0 or move > 8:
            print("Invalid number. Choose between 1 and 9.")
            continue
            
        if board[move] == ' ':
            board[move] = current_player
            
            if check_win(current_player):
                print_board()
                print(f"🏆 Player {current_player} wins!")
                break
                
            # Switch player
            current_player = 'O' if current_player == 'X' else 'X'
        else:
            print("Spot already taken! Try another one.")
            
    except ValueError:
        print("Please type a valid number.")

# If the loop finishes without a break, it's a tie
else:
    print_board()
    print("It's a draw! 🤝")
